def calc_bsa(height_cm, weight_kg):
    if not 80 <= height_cm <= 250:
        raise ValueError("身高须在80-250 cm范围内。")
    if not 20 <= weight_kg <= 350:
        raise ValueError("体重须在20-350 kg范围内。")
    return math.sqrt(height_cm * weight_kg / 3600.0)
